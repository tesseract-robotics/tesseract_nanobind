cmake_minimum_required(VERSION 3.20)
project(ompl_seed_boundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
find_package(ompl CONFIG REQUIRED)
add_library(seed_setter SHARED setter.cpp)
add_library(seed_reader SHARED reader.cpp)
target_link_libraries(seed_setter PRIVATE ompl::ompl)
target_link_libraries(seed_reader PRIVATE ompl::ompl)
add_executable(check_seed main.cpp)
target_link_libraries(check_seed PRIVATE seed_setter seed_reader)
